#include <bits/stdc++.h>
using namespace std;

signed main()
{
    int n; cin >> n;
    double S = 0, T = 0;
    for (int i = 1; i <= n; i++)
    {
        double gpa, sc;
        cin >> gpa >> sc;
        S += sc;
        T += gpa * sc;
    }
    if (S == 0) 
    {
        cout << "0.0" << endl;
    }
    else 
    {
        double ams = T / S;
        cout << fixed << setprecision(1) << ams << endl;
    }
    return 0;

#include<iostream>
using namespace std;
int main()
{
    int m, n;
    while (cin >> m >> n)
    {
        if (m < 100 || n > 999 || m > n)
        {
            cout << "Invalid Value." << endl;
            continue;
        }
        bool has = false; 
        for (int x = m; x <= n; x++)
        {
            int bai = x / 100;
            int shi = x % 100 / 10;
            int ge = x % 10;
            int sum = bai*bai*bai + shi*shi*shi + ge*ge*ge;

            if (sum == x)
            {
                cout << x << endl;
                has = true;
            }
        }
    }
    return 0;
}
