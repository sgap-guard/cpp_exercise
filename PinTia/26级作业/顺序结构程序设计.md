```cpp
#include <bits/stdc++.h>
using namespace std;

signed main()
{
    int t1, t2;
    cin >> t1 >> t2;

    int h1 = t1 / 100, m1 = t1 % 100;
    int h2 = t2 / 100, m2 = t2 % 100;

    int total1 = h1 * 60 + m1;
    int total2 = h2 * 60 + m2;
    int diff = total2 - total1;

    int hours = diff / 60;
    int minutes = diff % 60;

    cout << setfill('0') << setw(2) << hours << ":"
         << setw(2) << minutes << endl;

    return 0;
}
```
