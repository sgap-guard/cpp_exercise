#include <bits/stdc++.h>
using namespace std;

signed main()
{
    double a, b, c; cin >> a >> b >> c;
    double delta = b * b - 4 * a * c;
    if (fabs(a) < 1e-6 && fabs(b) < 1e-6 && fabs(c) < 1e-6)
    {
        cout << "Zero Equation" << endl;
        return 0;
    }
    if (fabs(a) < 1e-6 && fabs(b) < 1e-6)
    {
        cout << "Not An Equation" << endl;
        return 0;
    }
    if (fabs(a) < 1e-6)
    {
        double x = -c / b;
        if (fabs(x) < 1e-5) x = 0.0;
        cout << fixed << setprecision(2) << x << endl;
        return 0;
    }
    if (fabs(delta) < 1e-6)
    {
        double x = -b / (2 * a);
        if (fabs(x) < 1e-5) x = 0.0;
        cout << fixed << setprecision(2) << x << endl;
    }
    else if (delta > 0)
    {
        double x1 = (-b + sqrt(delta)) / (2 * a);
        double x2 = (-b - sqrt(delta)) / (2 * a);
        if (fabs(x1) < 1e-5) x1 = 0.0;
        if (fabs(x2) < 1e-5) x2 = 0.0;
        if (x1 < x2) swap(x1, x2);
        cout << fixed << setprecision(2) << x1 << endl;
        cout << fixed << setprecision(2) << x2 << endl;
    }
    else
    {
        double real = -b / (2 * a);
        double imag = sqrt(-delta) / (2 * a);
        if (fabs(real) < 1e-5) real = 0.0;
        if (fabs(imag) < 1e-5) imag = 0.0;
        imag = fabs(imag);
        cout << fixed << setprecision(2) << real << "+" << imag << "i" << endl;
        cout << fixed << setprecision(2) << real << "-" << imag << "i" << endl;
    }
    return 0;
}
