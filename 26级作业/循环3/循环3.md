{
    if (x < 2)

    }
    for (int i = 2; i * i <= x; i++)
    {
        if (x % i == 0)
        {
            return false;

    return true;
}

        {
            if (!first)
            {
                cout << " ";
            }
            cout << i;
            first = false;
        }
    }
    cout << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

bool isPrime(int n)
{
    if (n < 2)
    {
        return false;
    }
    int limit = (int)sqrt(n);
    for (int i = 2; i <= limit; i++)
    {
        if (n % i == 0)
        {
            return false;
        }
    }
    return true;
}

int main()
{
    int count = 0;
    for (int i = 100; i <= 200; i++)
    {
        if (isPrime(i))
        {
            cout << setw(5) << i;
            count++;

            if (count % 8 == 0)
            {
                cout << endl;
            }
        }
    }
    if (count % 8 != 0)
    {
        cout << endl;
    }
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int digitSum(int x)
{
    int sum = 0;
    while (x > 0)
    {
        sum += x % 10;
        x /= 10;
    }
    return sum;
}

int main()
{
    int n; cin >> n;
    int count = 0;
    for (int i = 1; i < 1000; i++)
    {
        if (digitSum(i) == n)
        {
            cout << setw(8) << i;
            count++;

            if (count % 6 == 0)
            {
                cout << endl;
            }
        }
    }
    if (count % 6 != 0)
    {
        cout << endl;
    }
    return 0;
}
