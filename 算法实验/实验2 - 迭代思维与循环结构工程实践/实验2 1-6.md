### 1 判断素数
```cpp
#include <iostream>
using namespace std;

bool isPrime(int x)
{
    if (x < 2)
    {
        return false;
    }
    for (int i = 2; i * i <= x; i++)
    {
        if (x % i == 0)
        {
            return false;
        }
    }
    return true;
}

signed main()
{
    int a; cin >> a;
    if (isPrime(a))
        cout << "YES" << endl;
    else
        cout << "NO" << endl;
    return 0;
}
```

### 3 2的n次方
```cpp
#include <bits/stdc++.h>
using namespace std;

signed main()
{
    long long x;
    while (cin >> x)
    {
        if (x < 0) x = -x;
        if (((x & (x - 1)) == 0) && x > 0)
        {
            long long n = __builtin_ctz(x);
            cout << n << endl;
        }
        else
        {
            cout << "No!" << endl;
        }
    }
    return 0;
}
```
`__builtin_ctz(x)` 是 GCC/G++ 编译器内置的一个极其高效的位运算函数。它的的全称是 Count Trailing Zeros（统计末尾零的个数）。

计算一个正整数 x 的二进制表示中，从最低位（最右边）开始，连续有多少个 0。
### 4 统计学生成绩 
```cpp
#include <bits/stdc++.h>
using namespace std;

signed main()
{
    int n, sum = 0; cin >> n;
    for (int i = 1; i <= n; i++)
    {
        int a; cin >> a;
        sum += a;
    }
    double avg = 1.0 * sum / n;
    cout << "average = " << fixed << setprecision(1) << avg << endl;
    return 0;
}
```
### 5 整数每位数求和
```cpp
#include <bits/stdc++.h>
using namespace std;

signed main()
{
    int a; cin >> a;
    bool is_neg = false;
    if (a < 0) 
    {
        a = -a;
        is_neg = true;
    }
    string s = to_string(a);
    int sum = 0;
    for (char c : s)
    {
        sum += c - '0';
    }
    if (is_neg == true) cout << "-";
    cout << sum << endl;
    return 0;
}
```
### 6 国王的金币
```cpp
#include <bits/stdc++.h>
using namespace std;

signed main()
{
    int n, day = 0, k = 1; cin >> n;
    long long sum = 0;
    while (day < n)
    {
        for (int i = 1; i <= k; i++)
        {
            day++;
            sum += k;
            if (day == n)
            {
                cout << sum << endl;
                break;
            }
        }
        k++;
    }
    return 0;
}
```
