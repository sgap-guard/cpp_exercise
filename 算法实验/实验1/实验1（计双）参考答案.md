### 7
```cpp
#include <bits/stdc++.h>
using namespace std;

int main()
{
    string a; long long b;
    cin >> a >> b;
    // 特判：如果被除数为0
    if (a == "0")
    {
        cout << "0 0" << endl;
        return 0;
    }
    // 1. 将大数字符串倒序存入 vector 中
    vector<int> A;
    for (int i = a.size() - 1; i >= 0; i--)
    {
        A.push_back(a[i] - '0');
    }
    // 2. 模拟除法过程
    vector<int> C; // 用来存储商，由于是顺位算的，此时 C 是正序的（高位在前）
    long long r = 0; // 余数
    // 从 A 的最高位开始遍历（即 vector 的末尾）
    for (int i = A.size() - 1; i >= 0; i--)
    {
        r = r * 10 + A[i]; // 当前被除数
        C.push_back(r / b); // 存入当前位的商
        r = r % b;          // 更新余数
    }
    // 3. 去除前导零
    // 因为我们是从高位开始算的，所以 C 的头部可能是一堆 0
    // 为了方便用 vector 的 pop_back 去零，我们先将 C 翻转，变成和 A 一样的倒序存储
    // (这一步也可以不翻转，直接用指针找第一个非零元素，但翻转更符合 vector 操作习惯)
    vector<int> res;
    for (int i = C.size() - 1; i >= 0; i--)
    {
        res.push_back(C[i]);
    }
    // 此时 res 是倒序的商，末尾是高位，去零只需要 pop_back
    while (res.size() > 1 && res.back() == 0)
    {
        res.pop_back();
    }
    // 4. 输出结果（倒序打印）
    for (int i = res.size() - 1; i >= 0; i--)
    {
        cout << res[i];
    }
    cout << " " << r << endl;
    return 0;
}
```

```cpp
    double t; int year;
    double rate, sum;

    if (year >= 5)
    {
        rate = 50;

    else
    {
        rate = 30;
    }
    if (t <= 40)
    {
        sum = t * rate;
    }
    else
    {
        sum = 40 * rate + (t - 40) * rate * 1.5;
    }
```
